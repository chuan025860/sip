CREATE DATABASE chyunnweb2;
USE chyunnweb2;

CREATE TABLE incident(
    incidentId INT NOT NULL AUTO_INCREMENT PRIMARY KEY, -- 事件編號，主鍵，自動遞增
    belongingType VARCHAR(20)NOT  NULL,                	-- 歸屬類型
    customer VARCHAR(20),                             	 -- 事件相關客戶，字串
    car VARCHAR(20) ,                                 	-- 事件相關車輛，字串
    reporter VARCHAR(20) NOT NULL,                     -- 反應人，字串
    task VARCHAR(30) NOT NULL,                         -- 工作事項，字串
    eventPurpose TEXT NOT NULL,	 		       -- 事件主旨	
    eventContent TEXT NOT NULL,                        -- 事件內容，長文字
    reportCount INT NOT NULL,                          -- 回報數，整數
    importance VARCHAR(20) NOT NULL,                   -- 重要性
    department VARCHAR(20) NOT NULL,                   -- 處理部門，字串
    handler VARCHAR(20) NOT NULL,                      -- 處理人，字串
    status VARCHAR(20) NOT NULL,                       -- 處理狀態
    lastReportTime DATETIME NOT NULL,                  -- 最後回報時間
    reportDate DATE  NOT NULL,                         -- 反應日期
    invalid BOOLEAN DEFAULT FALSE,                      -- 作廢
    creator VARCHAR(20) NOT NULL                        -- 製表人，字串
);

CREATE TABLE report(
    reportId INT NOT NULL AUTO_INCREMENT PRIMARY KEY, -- 回報編號，主鍵，自動遞增	
    incidentId INT NOT NULL,                          -- 外鍵，關聯到 incident 表
    reportTime DATETIME NOT NULL,                     -- 回報時間
    reporter VARCHAR(100) NOT NULL,                   -- 回報人
    progress VARCHAR(50) NOT NULL,                    -- 處理進度
    executionDate DATE,                               -- 執行日期
    executor VARCHAR(100) ,                           -- 執行人
    content TEXT ,                                    -- 內容
    CONSTRAINT fk_incident FOREIGN KEY (incidentId)   -- 設置外鍵約束
        REFERENCES incident(incidentId)               -- 關聯到 incident 表的 incidentId 欄位
        ON DELETE CASCADE                             -- 當 incident 被刪除時，自動刪除相關的報告
        ON UPDATE CASCADE                             -- 當 incidentId 更新時，自動更新報告
);


CREATE TABLE user (
  id INT  AUTO_INCREMENT PRIMARY KEY,
  loginId          VARCHAR(100)    NOT NULL,
  userId           VARCHAR(50),
  username          VARCHAR(255)    NOT NULL,
  password          VARCHAR(255)    NOT NULL,
  employee_no VARCHAR(50) NOT NULL   ,
  email             VARCHAR(255),
  department_id     INT,
  sub_unit_id       INT,
  sub_sub_unit_id   INT,
  position_id       INT,
  enabled           BOOLEAN DEFAULT TRUE,
CONSTRAINT fk_user_dept
    FOREIGN KEY (department_id)
    REFERENCES department (id)
    ON DELETE SET NULL
    ON UPDATE CASCADE,

  CONSTRAINT fk_user_subunit
    FOREIGN KEY (sub_unit_id)
    REFERENCES sub_unit (id)
    ON DELETE SET NULL
    ON UPDATE CASCADE,

  CONSTRAINT fk_user_subsub
    FOREIGN KEY (sub_sub_unit_id)
    REFERENCES sub_sub_unit (id)
    ON DELETE SET NULL
    ON UPDATE CASCADE,

  CONSTRAINT fk_user_position
    FOREIGN KEY (position_id)
    REFERENCES position (id)
    ON DELETE SET NULL
    ON UPDATE CASCADE
);   
      
      
CREATE TABLE user_roles (
  loginId VARCHAR(100) NOT NULL,
  role     VARCHAR(50)  NOT NULL,
  PRIMARY KEY(loginId, role),
  FOREIGN KEY(loginId) REFERENCES `user`id ON DELETE CASCADE
); 


CREATE TABLE incident_files (
    fileId   INT NOT NULL AUTO_INCREMENT PRIMARY KEY,    -- 圖片ID
    fileName VARCHAR(50) NOT NULL ,              -- 檔案名稱
    incidentId INT NOT NULL,                    -- 關聯的事件ID
    filePath VARCHAR(255) NOT NULL,              -- 檔案儲存路徑
    FOREIGN KEY (incidentId) REFERENCES incident(incidentId)  -- 外鍵關聯incident表
);

CREATE TABLE report_files (
    fileId   INT NOT NULL AUTO_INCREMENT PRIMARY KEY,    -- 圖片ID
    fileName VARCHAR(50) NOT NULL ,              -- 檔案名稱
    reportId INT NOT NULL,                     -- 關聯的回報 ID
    filePath VARCHAR(255) NOT NULL,            -- 檔案儲存路徑
    FOREIGN KEY (reportId) REFERENCES report(reportId)  -- 外鍵關聯 report 表
);
CREATE TABLE inventory_equipment (
    final_property_id VARCHAR(20) PRIMARY KEY,  -- 最終財產編號
    property_id  VARCHAR(20),                  -- 財產編號 
    management_id  VARCHAR(20),               -- 列管編號
    category VARCHAR(20), -- 類別
    name VARCHAR(255), -- 名稱
    remarks VARCHAR(255), -- 備註	
    size_serial VARCHAR(50), -- 尺寸_序號
    quantity_unit VARCHAR(10), -- 數量_單位
    purchase_date VARCHAR(20), -- 購買日期
    supplier VARCHAR(50), -- 購入廠商
    purchase_amount VARCHAR(10) , -- 購買金額
    tax VARCHAR(255) , -- 稅金
    location VARCHAR(20), -- 放置處
    custodian VARCHAR(10), -- 保管人
    contact_window VARCHAR(20), -- 聯繫窗口
    change_record  TEXT,  -- 異動記錄
    inventory_date VARCHAR(20),   -- 盤點日期

    inventory_in_progress  BOOLEAN DEFAULT FALSE   -- 表示是否在進行盤點中，預設為 FALSE

--     隨機清單IT
CREATE TABLE randomsampling_it (
    final_property_id   VARCHAR(20) PRIMARY KEY,  -- 最終財產編號
    state BOOLEAN DEFAULT FALSE,     -- 盤點狀態，預設為 FALSE
    inventory_in_progress  BOOLEAN DEFAULT FALSE   -- 表示是否在進行盤點中，預設為 FALSE
);
 
--      隨機清單總表
CREATE TABLE randomsampling_all	(
    asset_id VARCHAR(20) PRIMARY KEY,  -- 最終財產編號
    state BOOLEAN DEFAULT FALSE,     -- 盤點狀態，預設為 FALSE
    inventory_in_progress  BOOLEAN DEFAULT FALSE   -- 表示是否在進行盤點中，預設為 FALSE
);

CREATE TABLE general_catalog(
    asset_id VARCHAR(50) PRIMARY KEY,        -- 財產編號
    category VARCHAR(100),                   -- 類別
    name VARCHAR(100),                       -- 名稱
    model VARCHAR(100),                      -- 型號
    serial_or_size VARCHAR(100),             -- 尺寸/序號
    quantity_or_unit VARCHAR(100),           -- 數量/單位
    purchase_date VARCHAR(20),                      -- 購買日期
    supplier VARCHAR(100),                   -- 購入廠商
    purchase_price VARCHAR(10),           -- 購買金額
    tax VARCHAR(20),                      -- 稅金
    storage_location VARCHAR(100),           -- 放置處
    custodian VARCHAR(100),                  -- 保管人
    contact VARCHAR(100),                    -- 聯繫窗口
    change_log TEXT ,                         -- 異動記錄
    inventory_date VARCHAR(20),   -- 盤點日期
    state BOOLEAN DEFAULT FALSE,       -- 盤點狀態，預設為 FALSE
    inventory_in_progress  BOOLEAN DEFAULT FALSE     --  檢查是否有更新過
);



CREATE TABLE inventory_equipment_files (
    fileId VARCHAR(50) NOT NULL PRIMARY KEY,  -- 圖片ID，使用時間戳 + 隨機數（唯一）
    final_property_id VARCHAR(20) NOT NULL,                    -- 關聯的事件ID
    fileName  VARCHAR(255) NOT NULL,              -- 檔案儲存路徑
    FOREIGN KEY (final_property_id) REFERENCES inventory_equipment(final_property_id)  -- 外鍵關聯incident表
);

CREATE TABLE general_catalog_files (
    fileId VARCHAR(50) NOT NULL PRIMARY KEY,  -- 文件 ID（時間戳 + 隨機數，唯一）
    asset_id VARCHAR(50) NOT NULL,                     -- 關聯的回報 ID
    fileName  VARCHAR(255) NOT NULL,            -- 檔案儲存路徑
    FOREIGN KEY (asset_id) REFERENCES general_catalog(asset_id)  -- 外鍵關聯 report 表
);

-- 財產放置處
CREATE TABLE location (                     
     id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,  -- id 欄位設定為自動遞增並且是主鍵
    	name VARCHAR(255) NOT NULL                  -- name 欄位必須有值
);
-- 財產更新歷史
CREATE TABLE asset_history (
    id INT  AUTO_INCREMENT PRIMARY KEY,             -- 每條歷史紀錄的唯一ID
    date VARCHAR(255) NOT NULL,                     -- 日期
    custodian VARCHAR(255) NOT NULL,                -- 保管人
    location VARCHAR(255) NOT NULL,                 -- 放置處
    change_record TEXT NOT NULL,                    -- 異動記錄
    inventory_equipment_final_property_id VARCHAR(255) , -- 外鍵，與 `inventory_equipment` 關聯
    FOREIGN KEY (inventory_equipment_final_property_id) REFERENCES inventory_equipment(final_property_id)
        ON DELETE CASCADE  -- 若 `inventory_equipment` 刪除，相關的 `AssetHistory` 也會刪除
);

-- 總表財產更新歷史
CREATE TABLE asset_history_All (
    id INT AUTO_INCREMENT PRIMARY KEY,                 -- 每條歷史紀錄的唯一ID
    date VARCHAR(255) NOT NULL,                        -- 日期
    custodian VARCHAR(255) NOT NULL,                   -- 保管人
    location VARCHAR(255) NOT NULL,                    -- 放置處
    change_record TEXT NOT NULL,                       -- 異動記錄
    asset_id VARCHAR(50),                              -- 外鍵，對應 general_catalog 的 asset_id
    FOREIGN KEY (asset_id) REFERENCES general_catalog(asset_id)
        ON DELETE CASCADE                              -- 若財產刪除，歷史紀錄也會刪除
);

-- 資源借用請求
CREATE TABLE resource_borrow_request (
    id INT AUTO_INCREMENT PRIMARY KEY,              -- 資源借用主鍵
    content VARCHAR(255) NOT NULL,                 -- 內容（必填）
    start_time DATETIME NOT NULL,                  -- 開始時間（必填）
    end_time DATETIME NOT NULL,                    -- 結束時間（必填）
    category VARCHAR(100) NOT NULL,                -- 類別（必填，ex: 會議室）
    item VARCHAR(255) NOT NULL,                    -- 項目（必填）
    location VARCHAR(255),                         -- 位置
    note TEXT,                                     -- 備註
    creator VARCHAR(100) NOT NULL,                 -- 建立者（關聯 user.loginId）
    participants TEXT,                             -- 參與人員
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP, -- 建立時間
    CONSTRAINT fk_creator FOREIGN KEY (creator) REFERENCES user(loginId)
);

 -- 部門
CREATE TABLE department (
  id   INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100) NOT NULL UNIQUE
);

-- 小課／室（可以只有部門沒有子單位，就不塞資料）
CREATE TABLE sub_unit (
  id            INT AUTO_INCREMENT PRIMARY KEY,
  department_id INT NOT NULL,
  name VARCHAR(100) NOT NULL,
  UNIQUE(department_id, name),
  FOREIGN KEY (department_id) REFERENCES department(id)
);

CREATE TABLE sub_sub_unit (
  id            INT AUTO_INCREMENT PRIMARY KEY,
  sub_unit_id   INT NOT NULL,
  name          VARCHAR(100) NOT NULL,
  UNIQUE KEY uk_ssu_subunit_name (sub_unit_id, name),
  FOREIGN KEY (sub_unit_id) REFERENCES sub_unit(id)
);

-- 職位
CREATE TABLE position (
  id         INT AUTO_INCREMENT PRIMARY KEY,
  department_id INT NOT NULL,
  sub_unit_id   INT NULL,
  sub_sub_unit_id INT NULL,
  name VARCHAR(100) NOT NULL,
  UNIQUE(department_id, sub_unit_id, name),
  FOREIGN KEY (department_id) REFERENCES department(id),
  FOREIGN KEY (sub_unit_id)   REFERENCES sub_unit(id),
  FOREIGN KEY (sub_sub_unit_id) REFERENCES sub_sub_unit(id)
);
